cmake_minimum_required(VERSION 3.21)
project(YouCam VERSION 2.0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_include_directories(core_tests PRIVATE src)
add_test(NAME core COMMAND core_tests)
if(WIN32)
    enable_language(RC)
    add_library(youcam_visuals STATIC src/visuals.cpp)
    target_include_directories(youcam_visuals PUBLIC src)
    target_compile_definitions(youcam_visuals PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
    target_link_libraries(youcam_visuals PUBLIC gdiplus user32 gdi32)
    add_library(youcam_native STATIC src/camera.cpp src/audio.cpp src/capture.cpp src/recorder.cpp)
    target_include_directories(youcam_native PUBLIC src)
    target_compile_definitions(youcam_native PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
    target_link_libraries(youcam_native PUBLIC mfplat mfreadwrite mfuuid mf ole32 propsys shell32 user32 gdi32 dwmapi d3d11 dxgi windowsapp runtimeobject comctl32)
    target_link_libraries(youcam_native PUBLIC youcam_visuals)
    add_executable(visual_tests tests/visual_tests.cpp)
    target_link_libraries(visual_tests PRIVATE youcam_visuals)
    add_test(NAME visuals COMMAND visual_tests)
    add_executable(YouCam WIN32 src/main.cpp src/app.cpp src/app_visuals.cpp src/youcam.rc)
    target_include_directories(YouCam PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(YouCam PRIVATE youcam_native)
    target_link_options(YouCam PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/src/youcam.manifest")
    add_executable(app_tests tests/app_tests.cpp src/app.cpp src/app_visuals.cpp src/youcam.rc)
    target_include_directories(app_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(app_tests PRIVATE youcam_native)
    add_test(NAME interactions COMMAND app_tests)
    add_executable(native_tests tests/native_tests.cpp)
    target_link_libraries(native_tests PRIVATE youcam_native)
    add_executable(camera_check tests/camera_check.cpp)
    target_link_libraries(camera_check PRIVATE youcam_native)
    add_executable(sync_tests tests/sync_tests.cpp)
    target_link_libraries(sync_tests PRIVATE youcam_native)
    add_test(NAME sync COMMAND sync_tests)
    set_tests_properties(sync PROPERTIES TIMEOUT 30)
    foreach(target youcam_native youcam_visuals YouCam native_tests core_tests camera_check visual_tests app_tests sync_tests)
        target_compile_options(${target} PRIVATE /W4 /utf-8 /EHsc)
    endforeach()
    add_test(NAME native COMMAND native_tests)
    add_test(NAME ui COMMAND YouCam --smoke-test)
    set_tests_properties(native ui interactions PROPERTIES TIMEOUT 120 RUN_SERIAL TRUE)
else()
    message(STATUS "Non-Windows host: building the portable core checks only.")
endif()
