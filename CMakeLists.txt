cmake_minimum_required(VERSION 3.21)
project(YouCam VERSION 2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_include_directories(core_tests PRIVATE src)
add_test(NAME core COMMAND core_tests)
if(WIN32)
    add_library(youcam_native STATIC src/camera.cpp src/audio.cpp src/capture.cpp src/recorder.cpp)
    target_include_directories(youcam_native PUBLIC src)
    target_compile_definitions(youcam_native PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
    target_link_libraries(youcam_native PUBLIC mfplat mfreadwrite mfuuid mf ole32 propsys shell32 user32 gdi32 dwmapi d3d11 dxgi windowsapp runtimeobject comctl32)
    add_executable(YouCam WIN32 src/main.cpp src/app.cpp)
    target_link_libraries(YouCam PRIVATE youcam_native)
    target_link_options(YouCam PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/src/youcam.manifest")
    add_executable(native_tests tests/native_tests.cpp)
    target_link_libraries(native_tests PRIVATE youcam_native)
    foreach(target youcam_native YouCam native_tests core_tests)
        target_compile_options(${target} PRIVATE /W4 /utf-8 /EHsc)
    endforeach()
    add_test(NAME native COMMAND native_tests)
    add_test(NAME ui COMMAND YouCam --smoke-test)
    set_tests_properties(native ui PROPERTIES TIMEOUT 120)
else()
    message(STATUS "Non-Windows host: building the portable core checks only.")
endif()
